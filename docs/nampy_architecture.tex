  \node[gam box,text width=9.7cm,minimum height=1.05cm] (gam-entry) at (-7.25,-2.65)

     \scriptsize formula or array input; family and smooth specifications};
  \node[gam box,text width=9.7cm,minimum height=1.05cm] (gam-spec) at (-7.25,-4.65)
    {\textbf{\texttt{formula/} + \texttt{specs/}}\\[-1pt]
     \scriptsize parse predictors, terms, \texttt{by}/\texttt{id} links, and families};
  \node[gam box,text width=9.7cm,minimum height=1.10cm] (gam-basis) at (-7.25,-6.75)
    {\textbf{\texttt{smooths/} + \texttt{splines/} + \texttt{constraints/}}\\[-1pt]
     \scriptsize runtime terms own basis semantics; build ordinary or SCOP bases,\\
     penalties, identifiability constraints, and coefficient maps};
  \node[gam box,text width=9.7cm,minimum height=1.05cm] (gam-compile) at (-7.25,-8.90)
    {\textbf{\texttt{compiler/}}\\[-1pt]
     \scriptsize \texttt{CompiledTerm} $\rightarrow$ \texttt{CompiledPredictor} $\rightarrow$ \texttt{CompiledModel}};
  \node[gam box,text width=5.15cm,minimum height=2.05cm] (mgcv-route) at (-10.35,-11.65)
    {\textbf{ordinary \texttt{mgcv}-aligned route}\\[4pt]
     \scriptsize identity coefficient map\\
     Gaussian exact / PIRLS / general-family solvers\\
     REML, ML, GCV, UBRE selection};
  \node[scam box,text width=5.15cm,minimum height=2.05cm] (scam-route) at (-4.15,-11.65)
    {\textbf{shape-constrained \texttt{scam}-aligned route}\\[4pt]
     \scriptsize SCOP smooths + transformed coefficients\\
     constrained fitting iteration\\
     GCV/UBRE selection};
  \node[gam box,text width=9.7cm,minimum height=1.05cm] (gam-result) at (-7.25,-14.25)
    {\textbf{fitted statistical state}\\[-1pt]
     \scriptsize \texttt{GAMResult}, \texttt{FitCoreSolution}, covariance,\\
     EDF, smoothing parameters, diagnostics};
  \node[gam box,text width=9.7cm,minimum height=1.05cm] (gam-output) at (-7.25,-16.35)
    {\textbf{\texttt{predict/} + \texttt{inference/} + \texttt{diagnostics/}}\\[-1pt]
     \scriptsize prediction transforms remain paired with the fitted runtime terms};

  \node[neural box,text width=9.7cm,minimum height=1.05cm] (registry) at (7.25,-2.65)

  \node[neural box,text width=3.45cm,minimum height=1.55cm] (data) at (2.95,-5.15)

  \node[neural box,text width=3.45cm,minimum height=1.55cm] (architectures) at (7.25,-5.15)

  \node[neural box,text width=3.45cm,minimum height=1.55cm] (objectives) at (11.55,-5.15)

  \node[neural box,text width=3.45cm,minimum height=2.05cm] (fixed) at (7.25,-9.25)

  \node[neural box,text width=3.45cm,minimum height=2.05cm] (native) at (11.55,-9.25)

  \node[neural box,text width=9.7cm,minimum height=1.05cm] (neural-result) at (7.25,-12.25)

  \node[neural box,text width=9.7cm,minimum height=1.05cm] (neural-output) at (7.25,-16.35)

  \node[shared box,text width=18.5cm,minimum height=1.45cm] (shared) at (0,-19.25)

     \textbullet\quad \texttt{nampy.explanations}: tidy term tables\quad
     \textbullet\quad \texttt{nampy.plotting}: common renderer};
  \node[api box,text width=15.8cm,minimum height=1.15cm] (user-output) at (0,-21.45)
    {\textbf{common user workflow}\\[2pt]
     \scriptsize \texttt{fit} $\rightarrow$ \texttt{predict} $\rightarrow$ \texttt{score}
     $\rightarrow$ inspect terms $\rightarrow$ plot effects $\rightarrow$ persist fitted estimators};
